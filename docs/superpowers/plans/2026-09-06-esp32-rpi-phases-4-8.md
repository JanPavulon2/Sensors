# ESP32 <-> RPi5 Phases 4-8 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Take the ESP32 <-> RPi5 integration from "RPi hears heartbeats" (Phases 1-3, done) through real command delivery, a capability model, real addressable-LED rendering, multi-node sync, production hardening, and a new frontend "Nodes" page showing live connection state for every ESP32.

**Architecture:** The RPi5 side gets a `NodeRegistryService` that wraps the previously-unused `src/backend/core` `HostEngine`/`NodeInstance`/`NodeTransport` model (built in an earlier pass but never wired in) and folds today's standalone `api/routes/nodes.py` dict registry into it. A concrete `HttpNodeTransport` lets the Host push commands to ESP32s over HTTP; firmware gains a small `WebServer` to receive them. Remote LED zones (Phase 6) are a **separate, additive** pipeline from the existing local-GPIO `ZoneService`/`FrameManager` stack - they do not touch `ZoneID`, `zones.yaml`, `HardwareManager`, or `FrameManager`, because remote zones render on the ESP32 itself, not on the RPi's own strips. The frontend gets a new `features/nodes` module mirroring the existing `features/zones` realtime pattern (`useSyncExternalStore` + Socket.IO), plus a new top-level `/nodes` page.

**Tech Stack:** Python 3.9+/asyncio/FastAPI/Socket.IO (backend), React 18/TypeScript/Vite/Zustand-free `useSyncExternalStore` (frontend), Arduino/PlatformIO C++ with FastLED + ArduinoJson (firmware).

**Spec:** `docs/architecture-vnext/05-esp32-rpi-roadmap.md` (Phases 4-8), `docs/architecture-vnext/04-capabilities.md` (capability vocabulary), `docs/architecture-vnext/02-communication-and-event-flow.md` (event/command model).

## Global Constraints

- Never call hardware directly from a service - local zones still render only through `FrameManager`; remote (ESP32) zones render only through `NodeRegistryService`/`HttpNodeTransport`. Do not cross the two.
- Constructor injection only, no inline imports, full type hints, full words in identifiers (no abbreviations) - per `.claude/CLAUDE.md`.
- Never remove existing comments without instruction; update them if code they describe changes.
- All new REST endpoints are public (no auth), matching the existing `zones.py`/`nodes.py` convention - do not add auth middleware.
- Firmware changes must build clean via `pio run` (run from `firmware/firmware/`) before being considered done - there is no way to unit-test C++ in this repo, so "builds clean" + a described manual hardware check is the bar for firmware tasks.
- Backend Python tests run via `pytest` from the repo root (`pythonpath` covers `src`; this plan adds `src/backend` too - see Task 1).
- Frontend tests run via `npm test` (vitest, already configured with jsdom + `@testing-library/react`) from `frontend/`.

```python
# src/backend/core/capabilities.py
"""Capability vocabulary shared across node/device registration.

Mirrors docs/architecture-vnext/04-capabilities.md - keep both in sync.
"""

from __future__ import annotations

NODE_CAPABILITIES: frozenset[str] = frozenset(
    {
        "render_output",
        "input_capture",
        "telemetry_publish",
        "command_execution",
        "local_automation",
        "storage_local",
    }
)

DEVICE_CAPABILITIES: frozenset[str] = frozenset(
    {
        "switch_binary",
        "dimming",
        "color_control_rgb",
        "color_control_rgbw",
        "addressable_pixels",
        "sensor_read_temperature",
        "sensor_read_humidity",
        "sound_output",
        "display_output",
    }
)

ESP32_LED_STRIP_DEVICE_CAPABILITIES: list[str] = ["addressable_pixels", "color_control_rgb"]


def split_known_node_capabilities(capabilities: list[str]) -> tuple[list[str], list[str]]:
    """Splits a heartbeat's reported capabilities into (known, unknown) against
    NODE_CAPABILITIES, so callers can log/reject unrecognized entries without
    hard-failing the heartbeat."""
    known = [capability for capability in capabilities if capability in NODE_CAPABILITIES]
    unknown = [capability for capability in capabilities if capability not in NODE_CAPABILITIES]
    return known, unknown
```

```python
# src/backend/core/esp32_command_device.py
from __future__ import annotations

from core.device import Device
from core.message import Message
from core.port import Port, PortDirection


class Esp32CommandDevice(Device):
    """Addressable command endpoint standing in for a remote ESP32 node.

    Registered under the node's NodeInstance so HostEngine._deliver() routes
    connected messages through that node's NodeTransport instead of handling
    them locally (see HostEngine._deliver: local devices call
    handle_message() directly, remote ones go through transport.send()).
    handle_message() here is therefore never actually invoked in production -
    the real command execution happens in ESP32 firmware after
    NodeTransport.send() delivers the payload over HTTP. This class exists
    purely to give the Connection graph an addressable port pair.
    """

    def __init__(self, name: str = "esp32_command_device"):
        super().__init__(name)
        self.command_input_port: Port = self.add_port("cmd_in", PortDirection.INPUT)
        self.state_output_port: Port = self.add_port("state_out", PortDirection.OUTPUT)

    async def handle_message(self, port: Port, message: Message):
        return []
```

```python
# src/backend/core/node_command_router_device.py
from __future__ import annotations

from typing import Dict, Optional

from core.device import Device
from core.message import Message
from core.port import Port, PortDirection


class NodeCommandRouterDevice(Device):
    """Local fan-out point: one dedicated output port per registered node.

    HostEngine.connect() wires each port 1:1 to that node's
    Esp32CommandDevice input port. A single shared output port would
    broadcast to every connection from it (see HostEngine._handle), which is
    wrong for node-targeted commands - hence one port per node here.
    """

    def __init__(self, name: str = "node_command_router"):
        super().__init__(name)
        self._output_port_by_node_id: Dict[str, Port] = {}

    def add_target(self, node_external_id: str) -> Port:
        port = self.add_port(f"cmd_out_{node_external_id}", PortDirection.OUTPUT)
        self._output_port_by_node_id[node_external_id] = port
        return port

    def get_target_port(self, node_external_id: str) -> Optional[Port]:
        return self._output_port_by_node_id.get(node_external_id)

    async def handle_message(self, port: Port, message: Message):
        return []
```

```python
# src/backend/core/http_node_transport.py
from __future__ import annotations

import httpx

from core.node_instance import NodeInstance
from core.node_link import NodeMessage
from utils.logger import get_logger, LogCategory

log = get_logger().for_category(LogCategory.API)

HTTP_COMMAND_TIMEOUT_SECONDS = 3.0


class HttpNodeTransport:
    """Concrete NodeTransport that POSTs commands to a node's HTTP command
    server. Implements the `NodeTransport` Protocol (core/node_transport.py)
    under TransportProtocol.WEB_API - the RPi push side of Phase 4's "RPi5
    sends commands" (the ESP32 side is the WebServer added to
    firmware/firmware/src/main.cpp in Task 10).

    send() deliberately swallows delivery failures (logs and returns) rather
    than raising: it runs inside HostEngine's single shared message-queue
    consumer loop (core/event_bus.py's `run()`, no try/except of its own), so
    letting an exception escape here - e.g. one unreachable/offline ESP32 -
    would kill that loop forever, silently breaking command delivery to
    *every* node for the rest of the process's life. A dropped command to one
    node must never take down delivery to the others. (Discovered by a live
    boot test against an unreachable node address while executing this plan
    - the shared consumer task died silently with no error surfaced to the
    caller of send_command(), since HostEngine.emit() only enqueues.)
    """

    def __init__(self, shared_secret: str):
        self._shared_secret = shared_secret
        self._client = httpx.AsyncClient(timeout=HTTP_COMMAND_TIMEOUT_SECONDS)

    async def send(self, destination_node: NodeInstance, packet: NodeMessage) -> None:
        url = f"http://{destination_node.address}:{destination_node.port}/command"
        body = {
            "name": packet.message.name,
            "payload": packet.message.payload,
        }

        try:
            response = await self._client.post(
                url,
                json=body,
                headers={"X-Node-Secret": self._shared_secret},
            )
            response.raise_for_status()
        except httpx.HTTPError as error:
            log.warn(f"Failed to deliver command '{packet.message.name}' to {url}: {error}")

    async def close(self) -> None:
        await self._client.aclose()
```

```python
# src/services/node_registry_service.py
"""Node Registry Service - registers ESP32 heartbeats as HostEngine
NodeInstances and routes outbound commands to them over HTTP.

Owns the core.host_engine.HostEngine (src/backend/core) that models
nodes/devices/connections, bridging the app-wide EventBus/ServiceContainer
world with that generic routing engine. Replaces the old standalone
in-memory dict registry in api/routes/nodes.py (see
docs/architecture-vnext/05-esp32-rpi-roadmap.md Phase 4: "fold today's
standalone nodes.py registry into that model").
"""

from __future__ import annotations

import asyncio
import uuid
from datetime import datetime, timezone
from typing import Dict, List, Optional
from uuid import UUID

from core.capabilities import split_known_node_capabilities
from core.esp32_command_device import Esp32CommandDevice
from core.host_engine import HostEngine
from core.http_node_transport import HttpNodeTransport
from core.message import Message, MessageType
from core.node import Node, NodeType
from core.node_command_router_device import NodeCommandRouterDevice
from core.node_instance import NodeInstance
from models.domain.node import NodeSnapshot
from models.events import NodeCameOnlineEvent, NodeSnapshotUpdatedEvent, NodeWentOfflineEvent
from services.event_bus import EventBus
from utils.logger import get_logger, LogCategory

log = get_logger().for_category(LogCategory.API)

ONLINE_THRESHOLD_SECONDS = 15.0
WATCHDOG_POLL_INTERVAL_SECONDS = 5.0

_ESP32_NODE_TEMPLATE = Node(id=uuid.uuid4(), node_type=NodeType.ESP32, default_capabilities=[])


class NodeRegistryService:
    """Registers ESP32 heartbeats as HostEngine NodeInstances and routes
    outbound commands to them over HTTP."""

    def __init__(self, event_bus: EventBus, node_shared_secret: str):
        self.event_bus = event_bus
        self._transport = HttpNodeTransport(node_shared_secret)
        self._host_engine = HostEngine()
        self._router = NodeCommandRouterDevice()
        self._host_engine.register_device(self._router)

        self._instance_id_by_external_id: Dict[str, UUID] = {}
        self._last_seen_by_external_id: Dict[str, datetime] = {}
        self._online_by_external_id: Dict[str, bool] = {}
        self._firmware_by_external_id: Dict[str, str] = {}

        self._watchdog_task: Optional[asyncio.Task] = None

    async def run_engine_loop(self) -> None:
        """Drives HostEngine's internal message queue. Must run as a
        background task for send_command()/emit() to actually reach
        _deliver() - HostEngine.start() is an infinite consumer loop."""
        await self._host_engine.start()

    async def register_or_update_node(
        self,
        external_node_id: str,
        node_type: str,
        address: str,
        command_port: int,
        firmware: str,
        capabilities: List[str],
    ) -> NodeSnapshot:
        known_capabilities, unknown_capabilities = split_known_node_capabilities(capabilities)
        if unknown_capabilities:
            log.warn(
                f"Node '{external_node_id}' reported unrecognized capabilities: {unknown_capabilities}"
            )

        received_at = datetime.now(timezone.utc)
        self._last_seen_by_external_id[external_node_id] = received_at
        self._firmware_by_external_id[external_node_id] = firmware

        instance_id = self._instance_id_by_external_id.get(external_node_id)
        is_new_node = instance_id is None
        if is_new_node:
            instance_id = uuid.uuid4()
            self._instance_id_by_external_id[external_node_id] = instance_id

        node_instance = NodeInstance(
            id=instance_id,
            node_id=_ESP32_NODE_TEMPLATE.id,
            address=address,
            port=command_port,
            capabilities=known_capabilities,
        )
        # NodeInstance must be registered (HostEngine.register_node) BEFORE
        # register_device() below - register_device() raises ValueError if
        # its node_id isn't already a key in HostEngine.nodes, so a brand
        # new node's NodeInstance has to exist first even though this is its
        # first heartbeat.
        self._host_engine.register_node(node_instance, transport=self._transport)

        if is_new_node:
            self._register_new_node(instance_id, external_node_id)

        if not self._online_by_external_id.get(external_node_id, False):
            self._online_by_external_id[external_node_id] = True
            log.info(f"Node '{external_node_id}' came online")
            await self.event_bus.publish(
                NodeCameOnlineEvent(node_id=external_node_id, node_type=node_type, address=address)
            )

        snapshot = self._build_snapshot(external_node_id, node_instance)
        await self.event_bus.publish(NodeSnapshotUpdatedEvent(snapshot=snapshot))
        return snapshot

    def _register_new_node(self, instance_id: UUID, external_node_id: str) -> None:
        command_device = Esp32CommandDevice()
        self._host_engine.register_device(command_device, node_id=instance_id)

        router_output_port = self._router.add_target(external_node_id)
        self._host_engine.connect(router_output_port, command_device.command_input_port)

        log.info(f"Registered new node '{external_node_id}' as instance {instance_id}")

    def _build_snapshot(self, external_node_id: str, node_instance: NodeInstance) -> NodeSnapshot:
        return NodeSnapshot(
            instance_id=str(node_instance.id),
            external_node_id=external_node_id,
            node_type=_ESP32_NODE_TEMPLATE.node_type.name,
            address=node_instance.address,
            command_port=node_instance.port,
            firmware=self._firmware_by_external_id.get(external_node_id, "unknown"),
            capabilities=node_instance.capabilities,
            online=self._online_by_external_id.get(external_node_id, False),
            last_seen=self._last_seen_by_external_id[external_node_id],
        )

    def get_node(self, external_node_id: str) -> Optional[NodeSnapshot]:
        instance_id = self._instance_id_by_external_id.get(external_node_id)
        if instance_id is None:
            return None
        node_instance = self._host_engine.nodes[instance_id].node_instance
        return self._build_snapshot(external_node_id, node_instance)

    def list_nodes(self) -> List[NodeSnapshot]:
        return [self.get_node(external_node_id) for external_node_id in self._instance_id_by_external_id]

    async def send_command(self, external_node_id: str, command_name: str, payload: dict) -> None:
        if external_node_id not in self._instance_id_by_external_id:
            raise ValueError(f"Unknown node: {external_node_id}")

        router_output_port = self._router.get_target_port(external_node_id)
        if router_output_port is None:
            raise ValueError(f"No command route registered for node: {external_node_id}")

        await self._host_engine.emit(
            router_output_port,
            Message(type=MessageType.COMMAND, name=command_name, payload=payload),
        )

    def _is_stale(self, external_node_id: str, now: datetime) -> bool:
        last_seen = self._last_seen_by_external_id.get(external_node_id)
        if last_seen is None:
            return False
        return (now - last_seen).total_seconds() > ONLINE_THRESHOLD_SECONDS

    async def _run_watchdog(self) -> None:
        while True:
            await asyncio.sleep(WATCHDOG_POLL_INTERVAL_SECONDS)
            now = datetime.now(timezone.utc)

            for external_node_id in list(self._instance_id_by_external_id):
                if self._online_by_external_id.get(external_node_id, False) and self._is_stale(external_node_id, now):
                    self._online_by_external_id[external_node_id] = False
                    log.warn(
                        f"Node '{external_node_id}' went offline (no heartbeat for over {ONLINE_THRESHOLD_SECONDS:.0f}s)"
                    )

                    node_instance = self._host_engine.nodes[self._instance_id_by_external_id[external_node_id]].node_instance
                    await self.event_bus.publish(
                        NodeWentOfflineEvent(
                            node_id=external_node_id,
                            node_type=_ESP32_NODE_TEMPLATE.node_type.name,
                            address=node_instance.address,
                        )
                    )
                    await self.event_bus.publish(NodeSnapshotUpdatedEvent(snapshot=self.get_node(external_node_id)))

    def start_watchdog(self) -> None:
        if self._watchdog_task is None:
            self._watchdog_task = asyncio.create_task(self._run_watchdog())
            log.info("Node registry watchdog started")

    async def stop_watchdog(self) -> None:
        if self._watchdog_task is None:
            return
        self._watchdog_task.cancel()
        try:
            await self._watchdog_task
        except asyncio.CancelledError:
            pass
        self._watchdog_task = None
        log.info("Node registry watchdog stopped")

    async def close(self) -> None:
        await self._transport.close()
```

```cpp
void beginIdentifyBlink() {
    identifyActive = true;
    identifyStartTime = millis();
    lastIdentifyToggleTime = millis();
    identifyLedIsOn = false;
    Serial.println("Identify command received - blinking");
}

// Takes priority over the normal breathe/zone-state rendering while active -
// loop() only calls updateBreatheAnimation() when this returns false.
bool updateIdentifyBlink() {
    if (!identifyActive) {
        return false;
    }

    unsigned long now = millis();
    if (now - identifyStartTime >= IDENTIFY_DURATION_MILLISECONDS) {
        identifyActive = false;
        return false;
    }

    if (now - lastIdentifyToggleTime >= IDENTIFY_BLINK_INTERVAL_MILLISECONDS) {
        lastIdentifyToggleTime = now;
        identifyLedIsOn = !identifyLedIsOn;
        ledcWrite(PWM_CHANNEL, identifyLedIsOn ? 255 : 0);
    }
    return true;
}

void handleCommandRequest() {
    if (commandServer.method() != HTTP_POST) {
        commandServer.send(405, "application/json", "{\"error\":\"method not allowed\"}");
        return;
    }

    String requestBody = commandServer.arg("plain");
    JsonDocument commandDocument;
    DeserializationError parseError = deserializeJson(commandDocument, requestBody);
    if (parseError) {
        commandServer.send(400, "application/json", "{\"error\":\"invalid json\"}");
        return;
    }

    String commandName = commandDocument["name"] | "";

    if (commandName == "identify") {
        beginIdentifyBlink();
    } else {
        // Unknown command names (including "zone.state" before Task 15 adds
        // its handler) are rejected rather than silently ignored, so a
        // misconfigured RPi command gets a visible error instead of nothing.
        commandServer.send(400, "application/json", "{\"error\":\"unknown command\"}");
        return;
    }

    commandServer.send(200, "application/json", "{\"status\":\"ok\"}");
}

void beginCommandServer() {
    commandServer.on("/command", HTTP_POST, handleCommandRequest);
    commandServer.begin();
    Serial.print("Command server listening on port ");
    Serial.println(NODE_COMMAND_PORT);
}
```

## Part C — Phase 6: Real addressable LEDs on ESP32 (stream state, not frames)

**Architecture note:** Remote zones are intentionally **not** modeled as `ZoneID`/`ZoneConfig`/`ZoneService` entries. That pipeline (`zones.yaml`, `zone_mapping.yaml`, `HardwareManager`, `FrameManager.register_led_channel`) exists to drive LED strips wired directly to the RPi's own GPIO pins at 60 FPS. An ESP32-driven strip is physically wired to the ESP32, not the RPi - the RPi only needs to push occasional state changes (color/brightness/on-off), and the ESP32 renders them locally. Reusing `ZoneID` would mean threading a "this zone has no GPIO, don't register it with FrameManager" special case through `ConfigManager`/`HardwareManager`/`LedChannelFactory` - all working, sensitive code with no test coverage for this plan to risk. Instead, remote zones get their own small, additive, independently-loaded config and service, per the roadmap's own "stream state, not frames" verdict.

```python
"""Remote zone domain models (Phase 6 - LED zones rendered on ESP32 nodes,
not on the RPi's own GPIO strips - see Part C's architecture note in the
esp32-rpi-phases-4-8 implementation plan)."""

from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import List, Optional

import yaml

from models.color import Color


@dataclass(frozen=True)
class RemoteZoneConfig:
    """Immutable remote zone configuration from esp32_zones.yaml"""
    node_id: str
    zone_label: str
    display_name: str
    pixel_count: int
    sync_group: Optional[str] = None


@dataclass
class RemoteZoneState:
    """Mutable remote zone state, mirrored to the owning ESP32 via
    RemoteZoneRenderBridge/NodeRegistryService."""
    color: Color
    brightness: int = 100
    is_on: bool = True
    render_mode: str = "STATIC"  # "STATIC" or "BREATHE" - matches firmware's built-in animations


def load_remote_zone_configs(config_path: Path) -> List[RemoteZoneConfig]:
    """Loads esp32_zones.yaml independently of ConfigManager's include
    system, since remote zones are a separate concern from the local
    GPIO-strip pipeline ConfigManager/HardwareManager own."""
    if not config_path.exists():
        return []

    with open(config_path, "r", encoding="utf-8") as config_file:
        raw_data = yaml.safe_load(config_file) or {}

    return [
        RemoteZoneConfig(
            node_id=entry["node_id"],
            zone_label=entry["zone_label"],
            display_name=entry.get("display_name", entry["zone_label"]),
            pixel_count=entry.get("pixel_count", 1),
            sync_group=entry.get("sync_group"),
        )
        for entry in raw_data.get("remote_zones", [])
    ]
```

```python
# src/services/remote_zone_render_bridge.py
"""Remote Zone Render Bridge - forwards remote-zone state changes to the
owning ESP32 node via NodeRegistryService.send_command(), instead of through
FrameManager (which only drives locally-wired GPIO strips - see
docs/architecture-vnext/05-esp32-rpi-roadmap.md Phase 6: "RPi streams zone
state, ESP32 renders it locally")."""

from __future__ import annotations

from pathlib import Path
from typing import Dict, List

from models.color import Color
from models.domain.remote_zone import RemoteZoneConfig, RemoteZoneState, load_remote_zone_configs
from services.node_registry_service import NodeRegistryService
from utils.logger import get_logger, LogCategory

log = get_logger().for_category(LogCategory.ZONE)


class RemoteZoneRenderBridge:
    def __init__(self, node_registry_service: NodeRegistryService, config_path: Path):
        self.node_registry_service = node_registry_service
        self._configs_by_label: Dict[str, RemoteZoneConfig] = {
            config.zone_label: config for config in load_remote_zone_configs(config_path)
        }
        self._states_by_label: Dict[str, RemoteZoneState] = {
            label: RemoteZoneState(color=Color.from_rgb(255, 255, 255))
            for label in self._configs_by_label
        }

    def list_zones(self) -> List[RemoteZoneConfig]:
        return list(self._configs_by_label.values())

    def get_zones_for_node(self, node_id: str) -> List[RemoteZoneConfig]:
        return [config for config in self._configs_by_label.values() if config.node_id == node_id]

    def get_state(self, zone_label: str) -> RemoteZoneState:
        return self._require_state(zone_label)

    async def set_color(self, zone_label: str, color: Color) -> None:
        state = self._require_state(zone_label)
        state.color = color
        await self._push_state(zone_label)

    async def set_brightness(self, zone_label: str, brightness: int) -> None:
        state = self._require_state(zone_label)
        state.brightness = max(0, min(100, brightness))
        await self._push_state(zone_label)

    async def set_is_on(self, zone_label: str, is_on: bool) -> None:
        state = self._require_state(zone_label)
        state.is_on = is_on
        await self._push_state(zone_label)

    def _require_state(self, zone_label: str) -> RemoteZoneState:
        if zone_label not in self._states_by_label:
            raise ValueError(f"Unknown remote zone: {zone_label}")
        return self._states_by_label[zone_label]

    async def _push_state(self, zone_label: str) -> None:
        config = self._configs_by_label[zone_label]
        state = self._states_by_label[zone_label]
        red, green, blue = state.color.to_rgb()

        await self.node_registry_service.send_command(
            config.node_id,
            "zone.state",
            {
                "is_on": state.is_on,
                "brightness": state.brightness,
                "color": {"r": red, "g": green, "b": blue},
                "render_mode": state.render_mode,
            },
        )
        log.info(f"Pushed state to remote zone '{zone_label}'", node_id=config.node_id)
```

```python
# src/services/multi_node_sync_broadcaster.py
"""Multi-Node Sync Broadcaster - periodically multicasts a shared phase
reference so ESP32 nodes in the same sync_group stay visually in lockstep
while running the built-in BREATHE animation (see
docs/architecture-vnext/05-esp32-rpi-roadmap.md Phase 7). Deliberately scoped
to phase-sync for that one animation - extend the payload if a future
animation needs the same treatment, don't generalize speculatively now."""

from __future__ import annotations

import asyncio
import json
import socket
import time
from typing import Dict, List

from services.node_registry_service import NodeRegistryService
from services.remote_zone_render_bridge import RemoteZoneRenderBridge
from utils.logger import get_logger, LogCategory

log = get_logger().for_category(LogCategory.SYSTEM)

SYNC_MULTICAST_GROUP = "239.255.10.10"
SYNC_MULTICAST_PORT = 41234
SYNC_BROADCAST_INTERVAL_SECONDS = 2.0
SYNC_CYCLE_DURATION_MILLISECONDS = 2400  # matches firmware's breathe full cycle


class MultiNodeSyncBroadcaster:
    def __init__(self, remote_zone_render_bridge: RemoteZoneRenderBridge, node_registry_service: NodeRegistryService):
        self.remote_zone_render_bridge = remote_zone_render_bridge
        self.node_registry_service = node_registry_service
        self._socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM, socket.IPPROTO_UDP)
        self._socket.setsockopt(socket.IPPROTO_IP, socket.IP_MULTICAST_TTL, 2)

    def compute_sync_groups(self) -> Dict[str, List[str]]:
        groups: Dict[str, List[str]] = {}
        for zone_config in self.remote_zone_render_bridge.list_zones():
            if zone_config.sync_group is None:
                continue

            node_snapshot = self.node_registry_service.get_node(zone_config.node_id)
            if node_snapshot is None or not node_snapshot.online:
                continue

            groups.setdefault(zone_config.sync_group, []).append(node_snapshot.address)
        return groups

    async def start(self) -> None:
        while True:
            await asyncio.sleep(SYNC_BROADCAST_INTERVAL_SECONDS)
            self._broadcast_sync_groups()

    def _broadcast_sync_groups(self) -> None:
        for sync_group_name, member_addresses in self.compute_sync_groups().items():
            if len(member_addresses) < 2:
                continue

            reference_epoch_milliseconds = int(time.time() * 1000)
            packet = json.dumps(
                {
                    "sync_group": sync_group_name,
                    "reference_epoch_ms": reference_epoch_milliseconds,
                    "cycle_duration_ms": SYNC_CYCLE_DURATION_MILLISECONDS,
                }
            ).encode("utf-8")

            self._socket.sendto(packet, (SYNC_MULTICAST_GROUP, SYNC_MULTICAST_PORT))
            log.debug(f"Broadcast sync reference for group '{sync_group_name}'", members=len(member_addresses))
```

```cpp
// NTP sync must follow a successful WiFi connection (mirrors
// beginMdnsIfNeeded()'s "runs once per connection" pattern) - the multicast
// join also needs a live interface.
void beginSyncListenerIfNeeded() {
    if (syncListenerStarted) {
        return;
    }

    configTime(0, 0, "pool.ntp.org");
    IPAddress multicastGroup;
    multicastGroup.fromString(SYNC_MULTICAST_GROUP_ADDRESS);
    if (syncUdp.beginMulticast(multicastGroup, SYNC_MULTICAST_PORT)) {
        syncListenerStarted = true;
        Serial.println("NTP + sync multicast listener started");
    }
}

unsigned long currentEpochMilliseconds() {
    time_t now;
    time(&now);
    return (unsigned long)now * 1000UL;
}

void updateSyncListener() {
    int packetSize = syncUdp.parsePacket();
    if (packetSize <= 0) {
        return;
    }

    char packetBuffer[256];
    int bytesRead = syncUdp.read(packetBuffer, sizeof(packetBuffer) - 1);
    if (bytesRead <= 0) {
        return;
    }
    packetBuffer[bytesRead] = '\0';

    JsonDocument syncDocument;
    if (deserializeJson(syncDocument, packetBuffer)) {
        return;
    }

    const char* syncGroup = syncDocument["sync_group"] | "";
    if (strcmp(syncGroup, NODE_SYNC_GROUP) != 0) {
        return;  // not our group - ignore
    }

    phaseReferenceEpochMilliseconds = syncDocument["reference_epoch_ms"] | 0;
    phaseCycleDurationMilliseconds = syncDocument["cycle_duration_ms"] | DEFAULT_BREATHE_CYCLE_DURATION_MILLISECONDS;
    phaseReferenceKnown = true;
}

// Triangle wave over the shared cycle, driven by NTP-synced wall-clock time
// instead of this device's own millis() - the actual "lockstep" mechanism.
uint8_t computeSyncedBreatheBrightness() {
    unsigned long elapsedMilliseconds =
        (currentEpochMilliseconds() - phaseReferenceEpochMilliseconds) % phaseCycleDurationMilliseconds;
    unsigned long halfCycleMilliseconds = phaseCycleDurationMilliseconds / 2;

    if (elapsedMilliseconds < halfCycleMilliseconds) {
        return (uint8_t)map(elapsedMilliseconds, 0, halfCycleMilliseconds, 0, 255);
    }
    return (uint8_t)map(elapsedMilliseconds - halfCycleMilliseconds, 0, halfCycleMilliseconds, 255, 0);
}
```

```typescript
// frontend/src/features/nodes/realtime/nodes.store.ts
import { useSyncExternalStore, useCallback } from 'react';
import type { NodeSnapshot } from '@/shared/types/domain/node';

type NodeMap = Record<string, NodeSnapshot>;

let nodes: NodeMap = {};
let cachedNodesArray: NodeSnapshot[] = [];
const listeners = new Set<() => void>();

function notify() {
  listeners.forEach((l) => l());
}

export function setNodesSnapshot(snapshot: NodeSnapshot[]): void {
  nodes = Object.fromEntries(snapshot.map(n => [n.external_node_id, n]));
  cachedNodesArray = snapshot;
  notify();
}

export function updateNodeSnapshot(node: NodeSnapshot): void {
  nodes = { ...nodes, [node.external_node_id]: node };
  cachedNodesArray = Object.values(nodes);
  notify();
}

function subscribe(callback: () => void): () => void {
  listeners.add(callback);
  return () => listeners.delete(callback);
}

function getNodesSnapshot(): NodeSnapshot[] {
  return cachedNodesArray;
}

function getServerSnapshot(): NodeSnapshot[] {
  return [];
}

export function useNodes(): NodeSnapshot[] {
  return useSyncExternalStore(
    subscribe,
    getNodesSnapshot,
    getServerSnapshot
  );
}

export function useNode(externalNodeId: string): NodeSnapshot | undefined {
  const getSnapshot = useCallback(() => nodes[externalNodeId], [externalNodeId]);
  const getNodeServerSnapshot = useCallback(() => undefined, []);

  return useSyncExternalStore(
    subscribe,
    getSnapshot,
    getNodeServerSnapshot
  );
}
```

```tsx
// frontend/src/features/nodes/components/NodeList.tsx
import { useEffect, useState } from "react";
import { Card } from "@/shared/ui/card";
import { Button } from "@/shared/ui/button";
import { ChevronDown, ChevronRight } from "lucide-react";
import { useNodes, setNodesSnapshot } from "@/features/nodes/realtime/nodes.store";
import { useNodesQuery, useRemoteZonesQuery } from "@/features/nodes/api/queries";
import { NodeCard } from "./NodeCard";
import { RemoteZoneControls } from "./RemoteZoneControls";
import type { NodeSnapshot } from "@/shared/types/domain/node";

function NodeZonesSection({ node }: { node: NodeSnapshot }): JSX.Element {
  const { data: zones, isLoading } = useRemoteZonesQuery(node.external_node_id);

  if (isLoading) {
    return <div className="text-xs text-gray-400 pl-4">Loading zones...</div>;
  }

  if (!zones || zones.length === 0) {
    return <div className="text-xs text-gray-500 pl-4">No remote zones configured for this node</div>;
  }

  return (
    <div className="pl-4 space-y-2">
      {zones.map((zone) => (
        <RemoteZoneControls key={zone.zone_label} zone={zone} />
      ))}
    </div>
  );
}

function ExpandableNode({ node }: { node: NodeSnapshot }): JSX.Element {
  const [expanded, setExpanded] = useState(false);

  return (
    <div className="space-y-2">
      <div className="flex items-start gap-2">
        <Button
          variant="ghost"
          size="sm"
          className="h-7 w-7 p-0 mt-4"
          onClick={() => setExpanded((prev) => !prev)}
          aria-label={expanded ? "Collapse zones" : "Expand zones"}
        >
          {expanded ? <ChevronDown className="w-4 h-4" /> : <ChevronRight className="w-4 h-4" />}
        </Button>
        <div className="flex-1">
          <NodeCard node={node} />
        </div>
      </div>
      {expanded && <NodeZonesSection node={node} />}
    </div>
  );
}

export function NodeList(): JSX.Element {
  const nodes = useNodes();
  const { data: initialNodes, isLoading, error } = useNodesQuery();

  // Seed the realtime store from the initial REST fetch so the page isn't
  // empty while waiting for the first Socket.IO snapshot - same pattern
  // zones/hooks/useZones.ts uses for the zones feature.
  useEffect(() => {
    if (initialNodes) {
      setNodesSnapshot(initialNodes);
    }
  }, [initialNodes]);

  if (isLoading && nodes.length === 0) {
    return <Card className="p-8 text-center text-gray-400">Loading nodes...</Card>;
  }

  if (error) {
    return <Card className="p-8 text-center text-red-400">Failed to load nodes: {error.message}</Card>;
  }

  if (nodes.length === 0) {
    return <Card className="p-8 text-center text-gray-400">No nodes have connected yet</Card>;
  }

  return (
    <div className="space-y-3">
      {nodes.map((node) => (
        <ExpandableNode key={node.external_node_id} node={node} />
      ))}
    </div>
  );
}
```
